                  scheduleDaily(context, 102, 15, 30, "Sesi Sore (15:30 WIB)");
                  scheduleDaily(context, 103, 20, 0, "Sesi Malam (20:00 WIB)");
              }

              private static void scheduleDaily(Context context, int requestCode, int hour, int minute, String sessionLabel) {
                  AlarmManager am = (AlarmManager) context.getSystemService(Context.ALARM_SERVICE);
                  if (am == null) return;
                  Intent intent = new Intent(context, AlarmReceiver.class);
                  intent.putExtra("SESSION_LABEL", sessionLabel);
                  PendingIntent pi = PendingIntent.getBroadcast(
                      context, requestCode, intent,
                      PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE
                  );
                  Calendar cal = Calendar.getInstance();
                  cal.set(Calendar.HOUR_OF_DAY, hour);
                  cal.set(Calendar.MINUTE, minute);
                  cal.set(Calendar.SECOND, 0);
                  if (cal.getTimeInMillis() <= System.currentTimeMillis()) {
                      cal.add(Calendar.DAY_OF_YEAR, 1);
                  }
                  am.setRepeating(AlarmManager.RTC_WAKEUP, cal.getTimeInMillis(), AlarmManager.INTERVAL_DAY, pi);
              }
          }

// ============================================================================
// MONEV MAGANGHUB REMINDER — UNITY ANDROID APP (AUTO-UI TANPA SETUP SCENE)
// Cukup taruh file ini di folder Assets/Scripts/ pada project Unity apa saja!
// Otomatis membuat tampilan UI, alarm 12:00, 15:30, 20:00 WIB & kirim WA
// ke 62895394853580 begitu game/aplikasi dijalankan.
// ============================================================================

using System;
using System.Collections;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.EventSystems;
using UnityEngine.Networking;

public class MonevMagangHubUnityApp : MonoBehaviour
{
    public const string TARGET_WA = "62895394853580";
    public const string PORTAL_URL = "https://maganghub.kemnaker.go.id";

    private Text clockText;
    private Text countdownText;
    private Text statusSiangText;
    private Text statusSoreText;
    private Text statusMalamText;
    private InputField tokenInput;

    private readonly int[] hours = { 12, 15, 20 };
    private readonly int[] minutes = { 0, 30, 0 };
    private readonly string[] labels = { "Sesi Siang (12:00)", "Sesi Sore (15:30)", "Sesi Malam (20:00)" };
    private readonly string[] keys = { "siang_1200", "sore_1530", "malam_2000" };

    // Otomatis dijalankan oleh Unity saat aplikasi mulai tanpa perlu pasang ke GameObject manual
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void AutoBootstrap()
    {
        if (FindObjectOfType<MonevMagangHubUnityApp>() != null) return;
        GameObject host = new GameObject("MonevMagangHubUnityApp_Host");
        DontDestroyOnLoad(host);
        host.AddComponent<MonevMagangHubUnityApp>();
    }

    private void Awake()
    {
        Application.runInBackground = true;
        Screen.sleepTimeout = SleepTimeout.NeverSleep;
        BuildCompleteUnityUI();
        ScheduleAndroidNativeAlarms();
    }

    private void Update()
    {
        DateTime now = DateTime.Now;
        if (clockText != null)
        {
            clockText.text = now.ToString("HH:mm:ss") + " WIB";
        }

        // Hitung mundur ke jadwal berikutnya
        DateTime nextTarget = DateTime.MaxValue;
        string nextLabel = "";
        for (int i = 0; i < 3; i++)
        {
            DateTime candidate = new DateTime(now.Year, now.Month, now.Day, hours[i], minutes[i], 0);
            if (candidate <= now) candidate = candidate.AddDays(1);
            if (candidate < nextTarget)
            {
                nextTarget = candidate;
                nextLabel = labels[i];
            }

            // Cek apakah tepat waktunya alarm berbunyi
            string fireKey = "fired_" + now.ToString("yyyyMMdd") + "_" + keys[i];
            if (now.Hour == hours[i] && now.Minute == minutes[i] && now.Second < 5)
            {
                if (PlayerPrefs.GetInt(fireKey, 0) == 0)
                {
                    PlayerPrefs.SetInt(fireKey, 1);
                    PlayerPrefs.Save();
                    Handheld.Vibrate();
                    TriggerWhatsAppReminder(i);
                }
            }
        }

        if (countdownText != null && nextTarget != DateTime.MaxValue)
        {
            TimeSpan diff = nextTarget - now;
            countdownText.text = string.Format(" Berikutnya: {0} (-{1:D2}:{2:D2}:{3:D2})",
                nextLabel, (int)diff.TotalHours, diff.Minutes, diff.Seconds);
        }
    }

    public void TriggerWhatsAppReminder(int slotIndex)
    {
        string sesi = labels[slotIndex];
        string pesan = "Halo Peserta MagangHub! \n" +
                       "*PENGINGAT PRESENSI MONEV MAGANGHUB*\n\n" +
                       "Sekarang jadwal *" + sesi + "*. Jangan lupa mengisi presensi dan log harian Monev MagangHub Anda.\n\n" +
                       "Link Portal: " + PORTAL_URL;

        string savedToken = PlayerPrefs.GetString("fonnte_token", "").Trim();
        if (!string.IsNullOrEmpty(savedToken))
        {
            StartCoroutine(SendViaFonnteCoroutine(savedToken, pesan));
        }

        string waUrl = "https://wa.me/" + TARGET_WA + "?text=" + UnityWebRequest.EscapeURL(pesan);
        Application.OpenURL(waUrl);
    }

    private IEnumerator SendViaFonnteCoroutine(string token, string message)
    {
        WWWForm form = new WWWForm();
        form.AddField("target", TARGET_WA);
        form.AddField("message", message);

        using (UnityWebRequest req = UnityWebRequest.Post("https://api.fonnte.com/send", form))
        {
            req.SetRequestHeader("Authorization", token);
            yield return req.SendWebRequest();
        }
    }

    public void ToggleDone(int slotIndex)
    {
        string key = "done_" + DateTime.Now.ToString("yyyyMMdd") + "_" + keys[slotIndex];
        int current = PlayerPrefs.GetInt(key, 0);
        PlayerPrefs.SetInt(key, current == 1 ? 0 : 1);
        PlayerPrefs.Save();
        RefreshStatusLabels();
    }

    private void RefreshStatusLabels()
    {
        string today = DateTime.Now.ToString("yyyyMMdd");
        Text[] statusTexts = { statusSiangText, statusSoreText, statusMalamText };
        for (int i = 0; i < 3; i++)
        {
            if (statusTexts[i] == null) continue;
            bool done = PlayerPrefs.GetInt("done_" + today + "_" + keys[i], 0) == 1;
            statusTexts[i].text = done ? "STATUS: SUDAH PRESENSI ✓" : "STATUS: BELUM PRESENSI";
            statusTexts[i].color = done ? new Color(0.02f, 0.59f, 0.41f) : new Color(0.85f, 0.35f, 0.05f);
        }
    }

    // Menjadwalkan Alarm Android Native via AndroidJavaObject (Tanpa Android Studio)
    private void ScheduleAndroidNativeAlarms()
    {
#if UNITY_ANDROID && !UNITY_EDITOR
        try
        {
            using (AndroidJavaClass unityPlayer = new AndroidJavaClass("com.unity3d.player.UnityPlayer"))
            using (AndroidJavaObject activity = unityPlayer.GetStatic<AndroidJavaObject>("currentActivity"))
            using (AndroidJavaObject context = activity.Call<AndroidJavaObject>("getApplicationContext"))
            {
                // Menyimpan preferensi jadwal di SharedPreferences native Android
                using (AndroidJavaObject prefs = context.Call<AndroidJavaObject>("getSharedPreferences", "MonevReminderPrefs", 0))
                using (AndroidJavaObject editor = prefs.Call<AndroidJavaObject>("edit"))
                {
                    editor.Call<AndroidJavaObject>("putString", "target_wa", TARGET_WA).Call<bool>("commit");
                }
            }
        }
        catch (Exception e)
        {
            Debug.LogWarning("Info Native Android: " + e.Message);
        }
#endif
    }

    // Membangun seluruh UI secara otomatis dari kode (Tidak perlu buat Canvas manual di Scene)
    private void BuildCompleteUnityUI()
    {
        if (FindObjectOfType<EventSystem>() == null)
        {
            GameObject es = new GameObject("EventSystem");
            es.AddComponent<EventSystem>();
            es.AddComponent<StandaloneInputModule>();
            DontDestroyOnLoad(es);
        }

        GameObject canvasObj = new GameObject("MonevCanvas");
        DontDestroyOnLoad(canvasObj);
        Canvas canvas = canvasObj.AddComponent<Canvas>();
        canvas.renderMode = RenderMode.ScreenSpaceOverlay;

        CanvasScaler scaler = canvasObj.AddComponent<CanvasScaler>();
        scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(1080, 1920);
        scaler.matchWidthOrHeight = 0.5f;
        canvasObj.AddComponent<GraphicRaycaster>();

        Font font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
        if (font == null) font = Resources.GetBuiltinResource<Font>("Arial.ttf");

        // Background Panel
        GameObject bg = CreatePanel(canvasObj.transform, new Color(0.95f, 0.97f, 0.98f));
        VerticalLayoutGroup rootLayout = bg.AddComponent<VerticalLayoutGroup>();
        rootLayout.padding = new RectOffset(40, 40, 60, 40);
        rootLayout.spacing = 24;
        rootLayout.childControlWidth = true;
        rootLayout.childControlHeight = false;
        rootLayout.childForceExpandWidth = true;

        // Header Card
        GameObject headerCard = CreateCard(bg.transform, new Color(0.02f, 0.59f, 0.41f), 260);
        CreateLabel(headerCard.transform, font, "MONEV MAGANGHUB REMINDER", 40, FontStyle.Bold, Color.white);
        CreateLabel(headerCard.transform, font, "Target WA Otomatis: +" + TARGET_WA, 28, FontStyle.Normal, new Color(0.85f, 1f, 0.92f));
        clockText = CreateLabel(headerCard.transform, font, "12:00:00 WIB", 48, FontStyle.Bold, Color.white);
        countdownText = CreateLabel(headerCard.transform, font, "Menghitung jadwal berikutnya...", 26, FontStyle.Bold, new Color(1f, 0.92f, 0.5f));

        // 3 Schedule Cards (12:00, 15:30, 20:00)
        statusSiangText = BuildSessionCard(bg.transform, font, 0, "SESI 1 • JAM 12:00 SIANG", "Presensi & laporan tengah hari");
        statusSoreText  = BuildSessionCard(bg.transform, font, 1, "SESI 2 • JAM 15:30 SORE (SETENGAH 4)", "Presensi & pengisian log sore");
        statusMalamText = BuildSessionCard(bg.transform, font, 2, "SESI 3 • JAM 20:00 MALAM (JAM 8)", "Evaluasi & kelengkapan presensi malam");

        // Footer Portal & Fonnte Box
        GameObject footerCard = CreateCard(bg.transform, Color.white, 230);
        CreateLabel(footerCard.transform, font, "Opsional: Isi Token Fonnte.com agar WA terkirim otomatis di background", 24, FontStyle.Normal, new Color(0.3f, 0.35f, 0.4f));

        GameObject inputObj = new GameObject("TokenInput");
        inputObj.transform.SetParent(footerCard.transform, false);
        Image inputBg = inputObj.AddComponent<Image>();
        inputBg.color = new Color(0.93f, 0.95f, 0.97f);
        LayoutElement inputLE = inputObj.AddComponent<LayoutElement>();
        inputLE.preferredHeight = 64;
        tokenInput = inputObj.AddComponent<InputField>();
        Text inputText = CreateLabel(inputObj.transform, font, "", 26, FontStyle.Normal, Color.black);
        tokenInput.textComponent = inputText;
        tokenInput.text = PlayerPrefs.GetString("fonnte_token", "");
        tokenInput.onEndEdit.AddListener((val) => {
            PlayerPrefs.SetString("fonnte_token", val.Trim());
            PlayerPrefs.Save();
        });

        CreateButton(footerCard.transform, font, "BUKA PORTAL MONEV MAGANGHUB", new Color(0.08f, 0.12f, 0.2f), () => {
            Application.OpenURL(PORTAL_URL);
        });

        RefreshStatusLabels();
    }

    private Text BuildSessionCard(Transform parent, Font font, int index, string title, string subtitle)
    {
        GameObject card = CreateCard(parent, Color.white, 310);
        CreateLabel(card.transform, font, title, 34, FontStyle.Bold, new Color(0.08f, 0.12f, 0.2f));
        CreateLabel(card.transform, font, subtitle, 25, FontStyle.Normal, new Color(0.4f, 0.45f, 0.52f));
        Text statusLbl = CreateLabel(card.transform, font, "STATUS: BELUM PRESENSI", 26, FontStyle.Bold, new Color(0.85f, 0.35f, 0.05f));

        GameObject btnRow = new GameObject("BtnRow");
        btnRow.transform.SetParent(card.transform, false);
        HorizontalLayoutGroup hlg = btnRow.AddComponent<HorizontalLayoutGroup>();
        hlg.spacing = 16;
        hlg.childControlWidth = true;
        hlg.childControlHeight = true;
        hlg.childForceExpandWidth = true;
        LayoutElement rowLE = btnRow.AddComponent<LayoutElement>();
        rowLE.preferredHeight = 76;

        int captured = index;
        CreateButton(btnRow.transform, font, "TANDAI SUDAH ISI", new Color(0.02f, 0.59f, 0.41f), () => ToggleDone(captured));
        CreateButton(btnRow.transform, font, "KIRIM WA SEKARANG", new Color(0.15f, 0.22f, 0.32f), () => TriggerWhatsAppReminder(captured));

        return statusLbl;
    }

    private GameObject CreatePanel(Transform parent, Color color)
    {
        GameObject panel = new GameObject("Panel");
        panel.transform.SetParent(parent, false);
        RectTransform rt = panel.AddComponent<RectTransform>();
        rt.anchorMin = Vector2.zero;
        rt.anchorMax = Vector2.one;
        rt.offsetMin = Vector2.zero;
        rt.offsetMax = Vector2.zero;
        Image img = panel.AddComponent<Image>();
        img.color = color;
        return panel;
    }

    private GameObject CreateCard(Transform parent, Color bgColor, float height)
    {
        GameObject card = new GameObject("Card");
        card.transform.SetParent(parent, false);
        Image img = card.AddComponent<Image>();
        img.color = bgColor;
        LayoutElement le = card.AddComponent<LayoutElement>();
        le.preferredHeight = height;
        VerticalLayoutGroup vlg = card.AddComponent<VerticalLayoutGroup>();
        vlg.padding = new RectOffset(28, 28, 22, 22);
        vlg.spacing = 10;
        vlg.childControlWidth = true;
        vlg.childControlHeight = false;
        vlg.childForceExpandWidth = true;
        return card;
    }

    private Text CreateLabel(Transform parent, Font font, string content, int fontSize, FontStyle style, Color color)
    {
        GameObject txtObj = new GameObject("Label");
        txtObj.transform.SetParent(parent, false);
        Text txt = txtObj.AddComponent<Text>();
        txt.font = font;
        txt.text = content;
        txt.fontSize = fontSize;
        txt.fontStyle = style;
        txt.color = color;
        txt.alignment = TextAnchor.MiddleLeft;
        LayoutElement le = txtObj.AddComponent<LayoutElement>();
        le.preferredHeight = fontSize + 16;
        return txt;
    }

    private void CreateButton(Transform parent, Font font, string label, Color bgColor, UnityEngine.Events.UnityAction onClick)
    {
        GameObject btnObj = new GameObject("Button_" + label);
        btnObj.transform.SetParent(parent, false);
        Image img = btnObj.AddComponent<Image>();
        img.color = bgColor;
        Button btn = btnObj.AddComponent<Button>();
        btn.onClick.AddListener(onClick);
        LayoutElement le = btnObj.AddComponent<LayoutElement>();
        le.preferredHeight = 72;

        GameObject txtObj = new GameObject("Text");
        txtObj.transform.SetParent(btnObj.transform, false);
        RectTransform rt = txtObj.AddComponent<RectTransform>();
        rt.anchorMin = Vector2.zero;
        rt.anchorMax = Vector2.one;
        rt.offsetMin = Vector2.zero;
        rt.offsetMax = Vector2.zero;

        Text txt = txtObj.AddComponent<Text>();
        txt.font = font;
        txt.text = label;
        txt.fontSize = 26;
        txt.fontStyle = FontStyle.Bold;
        txt.color = Color.white;
        txt.alignment = TextAnchor.MiddleCenter;
    }
}
